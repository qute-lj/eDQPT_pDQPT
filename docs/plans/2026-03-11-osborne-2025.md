```julia
module DQPTOsborne2025

export finite_loschmidt_rate
export parse_osborne_cli

finite_loschmidt_rate(overlap::Number, L::Integer) = -log(abs2(overlap)) / L

function parse_osborne_cli(args::Vector{String})
    backend = :longrange
    length = 20
    steps = 20
    dt = 0.05
    output_prefix = "osborne2025"

    i = 1
    while i <= length(args)
        key = args[i]
        i == length(args) && throw(ArgumentError("missing value for $key"))
        value = args[i + 1]

        if key == "--backend"
            backend = Symbol(value)
        elseif key == "--length"
            length = parse(Int, value)
        elseif key == "--steps"
            steps = parse(Int, value)
        elseif key == "--dt"
            dt = parse(Float64, value)
        elseif key == "--output-prefix"
            output_prefix = value
        else
            throw(ArgumentError("unknown argument: $key"))
        end

        i += 2
    end

    return (backend = backend, length = length, steps = steps, dt = dt, output_prefix = output_prefix)
end

end
```

**Step 3: Write minimal implementation**

Add a product-state helper and a Hamiltonian backend switch. The first Hamiltonian attempt should be a finite long-range Ising chain constructed directly with `FiniteMPOHamiltonian` and pair couplings `J / |i-j|^alpha`; the fallback backend should keep only nearest-neighbor `zz` couplings plus a longitudinal field.

```julia
export build_polarized_state
export build_osborne_hamiltonian

function _product_tensor(amplitudes::AbstractVector{<:Number})
    data = reshape(ComplexF64.(collect(amplitudes)), 1, 2, 1)
    return TensorMap(data, ℂ^1 ⊗ ℂ^2 ← ℂ^1)
end

function build_polarized_state(L::Integer; up::Bool = true)
    amplitudes = up ? ComplexF64[1.0, 0.0] : ComplexF64[0.0, 1.0]
    return FiniteMPS([_product_tensor(amplitudes) for _ in 1:L])
end

function build_osborne_hamiltonian(
    backend::Symbol,
    L::Integer;
    J::Real = 1.0,
    g::Real = 0.7,
    alpha::Real = 3.0,
    hz::Real = 0.0,
)
    chain = fill(ℂ^2, L)

    local_x = Dict((i,) => -g * S_x for i in 1:L)
    local_z = Dict((i,) => -hz * S_z for i in 1:L)
    pair_ops = Dict{Tuple{Int, Int}, TensorMap}()

    for i in 1:(L - 1), j in (i + 1):L
        if backend == :longrange
            pair_ops[(i, j)] = -(J / abs(i - j)^alpha) * (S_z ⊗ S_z)
        elseif backend == :proxy
            j == i + 1 || continue
            pair_ops[(i, j)] = -J * (S_z ⊗ S_z)
        else
            throw(ArgumentError("unknown backend: $backend"))
        end
    end

    return FiniteMPOHamiltonian(chain, local_x) +
        FiniteMPOHamiltonian(chain, local_z) +
        FiniteMPOHamiltonian(chain, pair_ops)
end
```

Add a finite-chain quench driver that follows the finite MPSKit DQPT example and exports TSV output:

```julia
using Statistics: mean
using DelimitedFiles

export ensure_dir
export run_osborne_quench
export save_osborne_result

ensure_dir(path::AbstractString) = (mkpath(path); path)

function _mean_entropy(psi)
    return mean(real.(entropy(psi)))
end

function _mean_mz(psi, L::Integer)
    return mean(real(expectation_value(psi, i => S_z)) for i in 1:L)
end

function run_osborne_quench(;
    backend::Symbol = :proxy,
    length::Integer = 12,
    steps::Integer = 20,
    dt::Real = 0.05,
    J::Real = 1.0,
    g::Real = 0.7,
    alpha::Real = 3.0,
    hz::Real = 0.2,
    bond_dim::Integer = 16,
)
    psi0 = build_polarized_state(length; up = true)
    H = build_osborne_hamiltonian(backend, length; J = J, g = g, alpha = alpha, hz = hz)

    psi = deepcopy(psi0)
    envs = environments(psi, H)

    times = collect(0.0:Float64(dt):(Float64(dt) * steps))
    rate = Vector{Float64}(undef, length(times))
    entropy_vals = Vector{Float64}(undef, length(times))
    mz = Vector{Float64}(undef, length(times))

    for (k, _) in enumerate(times)
        rate[k] = finite_loschmidt_rate(dot(psi0, psi), length)
        entropy_vals[k] = _mean_entropy(psi)
        mz[k] = _mean_mz(psi, length)

        if k < length(times)
            alg = k <= 3 ? TDVP2(; trscheme = truncrank(bond_dim)) : TDVP()
            psi, envs = timestep(psi, H, 0, dt, alg, envs)
        end
    end

    return (times = times, rate = rate, entropy = entropy_vals, mz = mz)
end

function save_osborne_result(result, path::AbstractString)
    open(path, "w") do io
        println(io, "time\trate\tentropy\tmz")
        for i in eachindex(result.times)
            println(io, join((result.times[i], result.rate[i], result.entropy[i], result.mz[i]), '\t'))
        end
    end
    return path
end
```

```julia
export classify_dqpt_event
export summarize_osborne_regimes

function classify_dqpt_event(mz::AbstractVector{<:Real}, peak_index::Integer; tol::Real = 1e-6)
    window_lo = max(1, peak_index - 1)
    window_hi = min(length(mz), peak_index + 1)
    local_window = mz[window_lo:window_hi]

    if minimum(local_window) < -tol && maximum(local_window) > tol
        return :manifold
    elseif any(abs.(local_window) .<= tol)
        return :manifold
    else
        return :branch
    end
end

function summarize_osborne_regimes(results::Vector, labels::Vector{String}, path::AbstractString)
    open(path, "w") do io
        println(io, "# Osborne 2025 Regime Summary")
        for (label, result) in zip(labels, results)
            peak_index = argmax(result.rate)
            cls = classify_dqpt_event(result.mz, peak_index)
            println(io, "")
            println(io, "## " * label)
            println(io, "- peak_time = " * string(result.times[peak_index]))
            println(io, "- peak_rate = " * string(result.rate[peak_index]))
            println(io, "- peak_mz = " * string(result.mz[peak_index]))
            println(io, "- classification = " * string(cls))
        end
    end
    return path
end
```
